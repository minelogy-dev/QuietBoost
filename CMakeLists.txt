cmake_minimum_required(VERSION 3.24)
project(QuietBoost VERSION 2.2.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "QuietBoost requires Windows and the MSVC C++ toolchain.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /permissive-)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
set(WINDOWS_LIBS ole32 uuid user32 comctl32 shell32 propsys advapi32 avrt wintrust crypt32 bcrypt gdiplus)
add_executable(QuietBoost WIN32 src/app.cpp src/audio.cpp src/platform.cpp src/installer.cpp src/floating.cpp src/startup.cpp resources/app.rc resources/driver.rc resources/app.manifest)
target_include_directories(QuietBoost PRIVATE src third_party/vbcable)
target_link_libraries(QuietBoost PRIVATE ${WINDOWS_LIBS})
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT QuietBoost)
add_executable(QuietBoostTests WIN32 tests/core_tests.cpp src/platform.cpp src/floating.cpp src/startup.cpp)
target_include_directories(QuietBoostTests PRIVATE src)
target_link_libraries(QuietBoostTests PRIVATE ${WINDOWS_LIBS})
enable_testing()
add_test(NAME NativeCore COMMAND QuietBoostTests --headless)
install(TARGETS QuietBoost RUNTIME DESTINATION .)
